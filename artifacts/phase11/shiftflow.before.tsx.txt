const dateLabel = (date: Date, opts: Intl.DateTimeFormatOptions) => date.toLocaleDateString("sr-Latn-RS", { ...opts, timeZone: "Europe/Belgrade" });
const weekRangeLabel = (dates: Date[]) => {
  const start = dates[0], end = dates[6];
  const startMonth = dateLabel(start, { month: "long" });
  const endMonth = dateLabel(end, { month: "long" });
  const startYear = start.getUTCFullYear(), endYear = end.getUTCFullYear();
  if (startMonth === endMonth && startYear === endYear) return `${start.getUTCDate()}–${end.getUTCDate()}. ${endMonth} ${endYear}.`;
  return `${start.getUTCDate()}. ${startMonth}${startYear !== endYear ? ` ${startYear}.` : ""} – ${end.getUTCDate()}. ${endMonth} ${endYear}.`;
};

            {showHours && <div id="settings-hours" className="settings-index-target" tabIndex={-1}><HoursPanel key={`${authUser}:${cloudId}:hours-settings`} companyId={cloudId} workspace={w} readOnly={readOnly} flush={flush} settingsOnly /></div>}
          </section>
        </>}
        <section id="settings-access" className="settings-section" tabIndex={-1} aria-labelledby="settings-access-title">
          <div className="settings-section-heading"><h2 id="settings-access-title">Pristup</h2></div>
          <div className="settings-access"><section className="panel connection-panel"><span className="connection-icon"><RiLockLine size={24} /></span><div><h2>{cloudId ? "Nalog je povezan" : "Radite sa timom"}</h2><p className="muted">{cloudId ? "Rasporedi su sačuvani i dostupni vašem timu." : "Prijavite se za zajednički rad i deljenje rasporeda."}</p></div><Button variant="outline" onClick={() => { setModal("auth"); setFormError(""); }}>{cloudId ? "Moj nalog" : "Prijava i registracija"}<RiArrowRightSLine size={16} /></Button></section>{cloudId && <BasicAccess key={`${authUser}:${cloudId}:access`} companyId={cloudId} role={role} flush={flush} reconnect={connect} />}</div>
        </section>
      </>}

      </main></div>

    {modal === "generate" && <ModalFrame title="Nova nedelja, dobar plan." description="ShiftFlow usklađuje potrebe kafića sa dostupnošću i pravilima vašeg tima." close={() => !busy && setModal(null)}><div className="generation-visual"><RiSparkling2Line size={29} /><span>Vaš tim. Pametno raspoređen.</span><div><span>{active.length} zaposlenih</span><span>7 dana</span><span>{required} mesta</span></div></div><div className="generation-checklist"><p><RiCheckLine size={16} />Dostupnost i pozicije zaposlenih</p><p><RiCheckLine size={16} />Fond sati i minimum {w.rules.minRestHours} h odmora</p><p><RiCheckLine size={16} />Ravnomernija raspodela sati i vikenda</p></div><label className="check-row"><input type="checkbox" checked={keepLocked} onChange={e => setKeepLocked(e.target.checked)} /><span>Zadrži zaključane smene<small>{schedule.assignments.filter(a => a.locked).length} zaključanih smena u ovom rasporedu</small></span><RiLockLine size={17} /></label>{schedule.assignments.length > 0 && <p className="muted text-small">Otključane smene biće zamenjene novim nacrtom. Promenu možete poništiti.</p>}<div className="modal-actions"><Button variant="outline" disabled={busy} onClick={() => setModal(null)}>Otkaži</Button><Button disabled={busy || !active.length || readOnly} onClick={() => { setBusy(true); setTimeout(() => { try { const generated = generateSchedule(w, week, keepLocked); updateSchedule(generated, "Automatski generisan raspored"); setModal(null); setPage("Raspored"); const gaps = missingRequirements(generated.assignments, w.requirements).reduce((n, r) => n + r.missing, 0); notify(gaps ? `Nacrt je spreman. Preostalo je ${gaps} nepopunjenih mesta.` : "Raspored je spreman. Sve smene su popunjene."); } finally { setBusy(false); } }, 100); }}><RiSparkling2Line size={16} />{busy ? "Pravimo vaš raspored…" : "Generiši raspored"}</Button></div></ModalFrame>}

    {modal === "employee" && !plannerMode && <ModalFrame title={editEmployee ? "Uredi zaposlenog" : "Novi član tima"} description="Dodajte osnovne podatke i pozicije za raspoređivanje." close={() => setModal(null)}><form className="dialog-form" onSubmit={saveEmployee}><label>Ime i prezime<input name="name" autoFocus required minLength={2} maxLength={80} defaultValue={editEmployee?.name} placeholder="npr. Ana Petrović" /></label><fieldset><legend>Pozicije</legend><div className="position-options">{positions.map(p => <label key={p}><input type="checkbox" name="positions" value={p} defaultChecked={editEmployee?.positions.includes(p)} />{p}</label>)}</div></fieldset><label>Maksimalno sati nedeljno (opciono)<input type="number" name="maximum" min="1" max="168" defaultValue={editEmployee?.maxHours} placeholder="Bez posebnog limita" /></label><div className="form-columns"><label>Ciljani nedeljni fond sati<input type="number" name="hours" min="8" max="80" step="1" defaultValue={editEmployee?.targetHours ?? 40} required /></label><label>Željena smena<select name="preferred" defaultValue={editEmployee?.preferredShift ?? ""}><option value="">Bez preference</option>{w.templates.map(t => <option key={t.id} value={t.id}>{t.name}</option>)}</select></label></div><label className="check-row"><input type="checkbox" name="active" defaultChecked={editEmployee?.active ?? true} /><span>Aktivan član tima<small>Neaktivni zaposleni ne ulaze u automatsko generisanje.</small></span></label>{formError && <p className="form-error" role="alert">{formError}</p>}<div className="modal-actions"><Button type="button" variant="outline" onClick={() => setModal(null)}>Otkaži</Button><Button type="submit">Sačuvaj zaposlenog</Button></div></form></ModalFrame>}

    {modal === "shift" && editShift && <ModalFrame title={editShift.id ? "Uredi smenu" : "Dodaj smenu"} description="Brza promena, uz automatsku proveru pravila." close={() => setModal(null)}><form className="dialog-form" onSubmit={saveShift}><label>Zaposleni<select name="employeeId" value={editShift.employeeId} onChange={e => setEditShift(s => ({ ...s, employeeId: e.target.value, position: w.employees.find(p => p.id === e.target.value)?.positions[0] }))}>{active.map(e => <option key={e.id} value={e.id}>{e.name}</option>)}</select></label><div className="form-columns"><label>Dan<select name="day" defaultValue={editShift.day}>{DAY_NAMES.map((name, i) => <option key={name} value={i}>{name}</option>)}</select></label><label>Smena<select name="templateId" value={editShift.templateId} onChange={e => { const t = w.templates.find(t => t.id === e.target.value)!; setEditShift(v => ({ ...v, templateId: t.id, start: t.start, end: t.end })); }}>{w.templates.map(t => <option key={t.id} value={t.id}>{t.name} · {t.start}–{t.end}</option>)}</select></label></div><div className="form-columns"><label>Početak ove smene<input type="time" name="start" required value={editShift.start ?? w.templates.find(t => t.id === editShift.templateId)?.start ?? "07:00"} onChange={e => setEditShift(v => ({ ...v, start: e.target.value }))} /></label><label>Kraj ove smene<input type="time" name="end" required value={editShift.end ?? w.templates.find(t => t.id === editShift.templateId)?.end ?? "15:00"} onChange={e => setEditShift(v => ({ ...v, end: e.target.value }))} /></label></div><p className="muted">Kraj pre početka znači završetak narednog dana. Menja se samo ova dodela.</p><label>Pozicija<select name="position" key={editShift.employeeId} defaultValue={editShift.position}>{(w.employees.find(e => e.id === editShift.employeeId)?.positions ?? positions).map(p => <option key={p}>{p}</option>)}</select></label>{!plannerMode && <label>Interna napomena<textarea name="note" rows={2} maxLength={500} defaultValue={editShift.note} placeholder="Vidljiva samo menadžerima…" /></label>}<label className="check-row"><input type="checkbox" name="locked" defaultChecked={editShift.locked} /><span>Zaključaj smenu<small>Ostaje sačuvana prilikom ponovnog generisanja.</small></span><RiLockLine size={17} /></label>{formError && <p className="form-error" role="alert">{formError}</p>}<div className="modal-actions">{editShift.id && <Button type="button" variant="ghost" onClick={() => { const original = schedule.assignments.find(a => a.id === editShift.id); if (original) { setEditShift({ ...original, id: undefined, locked: false }); setFormError("Izaberite novog zaposlenog ili drugi termin, pa sačuvajte kopiju."); } }}>Dupliraj</Button>}{editShift.id && <button type="button" className="delete-button" onClick={() => { updateSchedule({ ...schedule, assignments: schedule.assignments.filter(a => a.id !== editShift.id) }, "Obrisana smena"); setModal(null); }}><RiDeleteBinLine size={16} />Obriši</button>}<Button type="button" variant="outline" onClick={() => setModal(null)}>Otkaži</Button><Button type="submit">Sačuvaj smenu</Button></div></form></ModalFrame>}

    {modal === "publish" && !plannerMode && <ModalFrame title={schedule.status === "published" ? "Raspored je objavljen" : "Spremni za novu nedelju?"} description="Pregledajte raspored pre nego što ga podelite sa timom." close={() => setModal(null)}><div className="publish-summary"><RiCalendar2Line size={30} /><h3>{dates[0].getUTCDate()}–{dates[6].getUTCDate()}. {dateLabel(dates[6], { month: "long", year: "numeric" })}</h3><p>{active.length} zaposlenih · {schedule.assignments.length} smena · {totalHours} sati</p></div>{blockers.length || missingCount ? <div className="form-error"><strong>Još malo do gotovog rasporeda.</strong><p>{missingCount} nepopunjenih mesta i {blockers.length} kršenja pravila. Otvorite proveru rasporeda da vidite detalje.</p><Button variant="outline" onClick={() => { setShowProblems(true); setModal(null); }}>Pregledaj probleme</Button></div> : <p className="success-text"><RiCheckLine size={17} />Sve smene su popunjene i nema blokirajućih konflikata.</p>}<p className="muted text-small">{cloudId ? "Naknadne izmene automatski ažuriraju objavljeni pregled." : "Objavljivanje označava raspored kao završen. Za deljenje linkom se prijavite."}</p><div className="modal-actions"><Button variant="outline" onClick={() => setModal(null)}>Zatvori</Button><Button disabled={!!blockers.length || !!missingCount || readOnly} onClick={async () => { try { if (!(await publishAfterFlush(flush, getLatest, setState, week))) { notify("Sačekajte da se izmene sačuvaju."); return; } setHasPublished(true); setModal(null); notify(cloudId ? "Raspored je objavljen i spreman za deljenje." : "Raspored je označen kao objavljen."); } catch (e) { notify((e as Error).message); } }}><RiCheckLine size={16} />Objavi raspored</Button></div></ModalFrame>}

    {modal === "share" && !plannerMode && <ModalFrame title="Podelite dobar plan." description="Zaposleni vide samo objavljene smene, imena i pozicije. Interne napomene ostaju privatne." close={() => setModal(null)}>{!cloudId ? <div className="share-notice"><RiShareForwardLine size={31} /><h3>Za privatni link se prijavite</h3><p>Za privatni link se prijavite. Raspored možete preuzeti kao sliku ili PDF.</p><Button onClick={() => { setModal("export"); }}>Preuzmi raspored<RiDownloadLine size={16} /></Button></div> : !hasPublished && schedule.status !== "published" ? <div className="share-notice"><RiInformationLine size={28} /><h3>Prvo objavite raspored</h3><p>Tim vidi samo objavljen raspored.</p><Button onClick={() => setModal("publish")}>Pregledaj i objavi</Button></div> : <div className="dialog-form"><label>Privatni link<input readOnly value={shareUrl} placeholder="Generišite link za ovaj raspored" /></label><Button disabled={busy} onClick={async () => { setBusy(true); try { if (!(await flush())) throw new Error("Sačekajte da se izmene sačuvaju."); const url = await createShare(cloudId, week); setShareUrl(url); notify("Novi link je spreman. Prethodni link je opozvan."); } catch (e) { notify((e as Error).message); } finally { setBusy(false); } }}><RiRefreshLine size={16} />{shareUrl ? "Opozovi i generiši novi link" : "Generiši privatni link"}</Button><Button variant="outline" onClick={async () => { try { await revokeShare(cloudId, week); setShareUrl(""); notify("Privatni link je opozvan."); } catch (e) { notify((e as Error).message); } }}>Opozovi link</Button><p className="muted">Novi link važi 60 dana. Otvoreni pregled proverava izmene na 30 sekundi.</p>{shareUrl && <Button variant="outline" onClick={() => { void navigator.clipboard.writeText(shareUrl).then(() => notify("Link je kopiran.")).catch(() => notify("Označite i kopirajte link iz polja.")); }}>Kopiraj link</Button>}</div>}</ModalFrame>}

    {modal === "export" && !plannerMode && <ModalFrame title="Raspored, spreman za tim." description="Izaberite format za preuzimanje. Interne napomene se ne prikazuju u izvozu." close={() => setModal(null)}><div className="export-options"><button disabled={busy} onClick={() => { setModal(null); setView("week"); setQuery(""); setPosition("all"); setTimeout(() => window.print(), 200); }}><RiCalendar2Line size={25} /><strong>PDF dokument</strong><span>Otvori štampu → Sačuvaj kao PDF</span><RiArrowRightUpLine size={18} /></button><button disabled={busy} onClick={() => { setView("week"); setQuery(""); setPosition("all"); setTimeout(() => void exportImage(), 150); }}><RiDownloadLine size={25} /><strong>{busy ? "Pripremamo sliku…" : "PNG slika"}</strong><span>Jasna slika za grupni razgovor</span><RiArrowRightUpLine size={18} /></button></div></ModalFrame>}

    {modal === "auth" && <ModalFrame title={cloudId ? "Vaš nalog" : "Dobro došli u ShiftFlow."} description={cloudId ? authUser : "Prijavite se ili otvorite poslovni prostor za svoj tim."} close={() => setModal(null)}>{cloudId ? <div className="dialog-form"><p className="success-text"><RiLockLine size={17} />Povezano sa firmom {w.name}.</p><Button variant="outline" onClick={async () => { try { await signOut(); setModal(null); } catch (e) { notify((e as Error).message); } }}><RiLogoutBoxRLine size={16} />Odjavi se</Button></div> : !getSupabase() ? <div className="share-notice"><RiLockLine size={28} /><h3>Demo je spreman za isprobavanje</h3><p>Prijava trenutno nije dostupna. Raspored možete isprobati ovde.</p><Button onClick={() => setModal(null)}>Nastavi u demo prostoru</Button></div> : <form className="dialog-form" onSubmit={async e => { e.preventDefault(); const f = new FormData(e.currentTarget); const submitter = (e.nativeEvent as SubmitEvent).submitter as HTMLButtonElement; const signup = submitter?.value === "signup"; setBusy(true); setFormError(""); try { const db = getSupabase()!; const email = String(f.get("email")); const password = String(f.get("password")); const result = signup ? await db.auth.signUp({ email, password, options: { emailRedirectTo: location.origin + (new URLSearchParams(location.search).has("invite") ? location.search : ""), data: { company_name: new URLSearchParams(location.search).has("invite") ? undefined : String(f.get("company") || "Moj kafić") } } }) : await db.auth.signInWithPassword({ email, password }); if (result.error) throw result.error; if (signup && !result.data.session) { notify("Proverite email da potvrdite registraciju, pa se prijavite."); setModal(null); } else { await connect(); setModal(null); notify("Uspešno ste prijavljeni."); } } catch (err) { setFormError((err as Error).message); } finally { setBusy(false); } }}><label>Email<input type="email" name="email" autoComplete="email" required /></label><label>Lozinka<input type="password" name="password" minLength={8} autoComplete="current-password" required /></label><label>Naziv firme <span className="muted">(za registraciju)</span><input name="company" placeholder="Moj kafić" maxLength={80} /></label>{formError && <p className="form-error">{formError}</p>}<Button disabled={busy} type="submit" value="signin">Prijavi se</Button><Button disabled={busy} variant="outline" type="submit" value="signup">Registruj firmu</Button></form>}</ModalFrame>}

    {modal === "notifications" && !plannerMode && <ModalFrame title="Obaveštenja" description="Obaveštenja za vaš objekat." close={() => setModal(null)}><BasicNotifications key={`${authUser}:${cloudId}:notifications`} companyId={cloudId} /></ModalFrame>}
    <ExportSchedule workspace={w} schedule={schedule} exportRef={exportRef} />
    {toast && <div className="toast" role="status"><RiCheckboxCircleLine size={19} /><span>{toast}</span><button className="icon-button" aria-label="Zatvori obaveštenje" onClick={() => setToast("")}><RiCloseLine size={16} /></button></div>}
  </div>;
